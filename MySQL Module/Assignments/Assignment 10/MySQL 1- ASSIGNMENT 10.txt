Q1) Write a stored function Annual_Salary that takes salary as input
and returns yearly salary (salary * 12).
Ans: delimiter #
CREATE FUNCTION Annual_Salary3(p_salary DECIMAL(10,2))
    -> RETURNS DECIMAL(12,2)
    -> DETERMINISTIC
    -> BEGIN
    ->     RETURN p_salary * 12;
    -> end #

Q2) Write a stored function Calculate_Bonus that takes salary as input
and returns bonus amount:
- If salary >= 80000 → bonus = 20%
- If salary between 50000 and 79999 → bonus = 10%
- Otherwise → bonus = 5%

Ans: delimiter #
CREATE FUNCTION Calculate_Bonus(p_salary DECIMAL(10,2))
    -> RETURNS DECIMAL(10,2)
    -> DETERMINISTIC
    -> BEGIN
    ->     IF p_salary >= 80000 THEN
    ->         RETURN p_salary * 0.20;
    ->     ELSEIF p_salary >= 50000 THEN
    ->         RETURN p_salary * 0.10;
    ->     ELSE
    ->         RETURN p_salary * 0.05;
    ->     END IF;
    -> end #
mysql> delimiter ;
mysql> select emp_id,emp_name,salary,

Q3) Write a stored function Salary_Grade that takes salary as input
and returns grade:
- 'A' if salary >= 80000
- 'B' if salary between 50000 and 79999
- 'C' if salary < 50000

Ans: delimiter #
CREATE FUNCTION Salary_Grade(p_salary DECIMAL(10,2))
    -> RETURNS CHAR(1)
    -> DETERMINISTIC
    -> BEGIN
    ->     IF p_salary >= 80000 THEN
    ->         RETURN 'A';
    ->     ELSEIF p_salary >= 50000 THEN
    ->         RETURN 'B';
    ->     ELSE
    ->         RETURN 'C';
    ->     END IF;
    -> END #
delimiter ;
